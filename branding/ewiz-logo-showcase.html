Ship it as a .icns

Master is vector — branding/ewiz-icon.svg. Rasterize once, then build the iconset:

rsvg-convert -w 1024 -h 1024 branding/ewiz-icon.svg > icon.png
mkdir eWiz.iconset
for s in 16 32 64 128 256 512; do
  sips -z $s $s        icon.png --out eWiz.iconset/icon_${s}x${s}.png
  sips -z $((s*2)) $((s*2)) icon.png --out eWiz.iconset/icon_${s}x${s}@2x.png
done
iconutil -c icns eWiz.iconset
